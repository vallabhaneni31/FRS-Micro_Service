# Python dependencies for face_quality_service.py (the face-quality sidecar).
#
# Install into the venv the PM2 processes run from:
#   cd backend/api/scripts
#   python3 -m venv venv
#   ./venv/bin/python3 -m pip install -r requirements.txt
#
# Model weights are NOT vendored and download on first run: InsightFace
# buffalo_sc into ~/.insightface/models, and the MediaPipe Face Landmarker
# bundle into ./models (see mediapipe_engine.py). Both need outbound HTTPS
# the first time the service starts on a new host.
#
# Versions pinned to what is running on the dev host as of 2026-08-28.

numpy==2.4.6

# InsightFace (buffalo_sc) — still required regardless of QUALITY_ENGINE:

onnxruntime==1.26.0

# MediaPipe engine (QUALITY_ENGINE=mediapipe) — enrollment quality/pose.
# scipy supplies the rotation-matrix -> Euler conversion for head pose.

# Python dependencies for vlm_quality_service.py (the VLM visitor-photo

# stack those other services use — a host that only runs face_quality_service.py
# shouldn't be forced to install this.
#
# Install into its own venv:
#   cd backend/api/scripts
#   python3 -m venv venv-vlm
#   ./venv-vlm/bin/python3 -m pip install -r requirements-vlm.txt
#
# Model weights are NOT vendored and download on first run (vikhyatk/moondream2
# from Hugging Face Hub) — needs outbound HTTPS the first time the service
# starts on a new host.
#
# Use the CPU build of torch unless the host actually has a CUDA GPU + drivers
